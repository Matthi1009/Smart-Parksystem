#include <SPI.h>
#include <MFRC522.h>
#include <ESP32Servo.h>

// ==========================================
//  PIN DEFINITIONEN (laut eurer Excel)
// ==========================================
#define RFID_SS     5    // RFID SDA/SS
#define RFID_RST    17   // RFID RST
#define TRIG_PIN    16   // Ultraschall Trigger
#define ECHO_PIN    13   // Ultraschall Echo (Spannungsteiler PFLICHT!)
#define LED_ROT     32   // Rote LED
#define LED_GRUEN   33   // Grüne LED
#define SERVO_PIN   26   // Servo SG90 Signal (ACHTUNG: IN4 vom Motor hier nicht belegen!)

// ==========================================
//  EINSTELLUNGEN
// ==========================================
#define MAX_PARKPLAETZE    10
#define SCHRANKE_AUF       90    // Grad offen
#define SCHRANKE_ZU        0     // Grad geschlossen
#define DISTANZ_AUTO       20    // cm – Auto erkannt
#define SCHRANKE_WARTEZEIT 3000  // ms – wie lange offen

// ==========================================
//  OBJEKTE
// ==========================================
MFRC522 rfid(RFID_SS, RFID_RST);
Servo schranke;

// ==========================================
//  GLOBALE VARIABLEN
// ==========================================
int parkplaetze = MAX_PARKPLAETZE;
bool ausfahrtAktiv = false;
unsigned long ausfahrtTimer = 0;

// Erlaubte RFID Karten – UIDs nach erstem Test eintragen!
String erlaubteKarten[] = {
  "5C:15:F4:38",   // Karte 1
  "FF:12:AB:44"    // Karte 2
};
int anzahlKarten = 2;

// ==========================================
//  SETUP
// ==========================================
void setup() {
  Serial.begin(115200);

  // Pins
  pinMode(TRIG_PIN,  OUTPUT);
  pinMode(ECHO_PIN,  INPUT);
  pinMode(LED_ROT,   OUTPUT);
  pinMode(LED_GRUEN, OUTPUT);

  // RFID
  SPI.begin();
  rfid.PCD_Init();

  // Servo
  schranke.attach(SERVO_PIN);
  schranke.write(SCHRANKE_ZU);
  delay(500);

  updateLEDs();

  Serial.println("=====================================");
  Serial.println("     Smart Parking gestartet         ");
  Serial.println("=====================================");
  Serial.print("Freie Plätze: ");
  Serial.print(parkplaetze);
  Serial.print(" / ");
  Serial.println(MAX_PARKPLAETZE);
  Serial.println();
}

// ==========================================
//  LOOP
// ==========================================
void loop() {
  checkEinfahrt();
  checkAusfahrt();

  // Schranke schließen nach Wartezeit
  if (ausfahrtAktiv && millis() - ausfahrtTimer > SCHRANKE_WARTEZEIT) {
    schliesseSchranke();
    ausfahrtAktiv = false;
    Serial.println("Schranke geschlossen.");
    Serial.println();
  }

  delay(100);
}

// ==========================================
//  EINFAHRT – RFID
// ==========================================
void checkEinfahrt() {
  if (!rfid.PICC_IsNewCardPresent()) return;
  if (!rfid.PICC_ReadCardSerial())   return;

  // UID lesen
  String uid = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) uid += "0";
    uid += String(rfid.uid.uidByte[i], HEX);
    if (i < rfid.uid.size - 1) uid += ":";
  }
  uid.toUpperCase();

  Serial.println("--- EINFAHRT ---");
  Serial.print("Karte: ");
  Serial.println(uid);

  if (karteErlaubt(uid)) {
    if (parkplaetze > 0) {
      parkplaetze--;
      Serial.println("Zugang OK – Auto fährt rein!");
      Serial.print("Noch frei: ");
      Serial.println(parkplaetze);
      oeffneSchranke();
      delay(SCHRANKE_WARTEZEIT);
      schliesseSchranke();
      updateLEDs();
    } else {
      Serial.println("VOLL – Einfahrt verweigert!");
      blinkeRot(3);
    }
  } else {
    Serial.println("Karte NICHT berechtigt!");
    blinkeRot(2);
  }

  Serial.println();
  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
}

// ==========================================
//  AUSFAHRT – ULTRASCHALL
// ==========================================
void checkAusfahrt() {
  if (ausfahrtAktiv) return;

  int distanz = getDistanz();

  if (distanz > 0 && distanz < DISTANZ_AUTO) {
    Serial.println("--- AUSFAHRT ---");
    Serial.print("Auto erkannt, Distanz: ");
    Serial.print(distanz);
    Serial.println(" cm");

    parkplaetze++;
    if (parkplaetze > MAX_PARKPLAETZE) parkplaetze = MAX_PARKPLAETZE;

    Serial.print("Freie Plätze: ");
    Serial.println(parkplaetze);

    oeffneSchranke();
    updateLEDs();
    ausfahrtAktiv = true;
    ausfahrtTimer = millis();
    Serial.println();
  }
}

// ==========================================
//  HILFSFUNKTIONEN
// ==========================================

int getDistanz() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long dauer = pulseIn(ECHO_PIN, HIGH, 30000);
  if (dauer == 0) return 999;
  return dauer * 0.034 / 2;
}

void oeffneSchranke() {
  Serial.println("Schranke öffnet...");
  schranke.write(SCHRANKE_AUF);
}

void schliesseSchranke() {
  schranke.write(SCHRANKE_ZU);
}

bool karteErlaubt(String uid) {
  for (int i = 0; i < anzahlKarten; i++) {
    if (erlaubteKarten[i] == uid) return true;
  }
  return false;
}

void updateLEDs() {
  if (parkplaetze <= 0) {
    digitalWrite(LED_GRUEN, LOW);
    digitalWrite(LED_ROT,   HIGH);
  } else {
    digitalWrite(LED_GRUEN, HIGH);
    digitalWrite(LED_ROT,   LOW);
  }
}

void blinkeRot(int mal) {
  for (int i = 0; i < mal; i++) {
    digitalWrite(LED_ROT, HIGH);
    delay(200);
    digitalWrite(LED_ROT, LOW);
    delay(200);
  }
  updateLEDs();
}
